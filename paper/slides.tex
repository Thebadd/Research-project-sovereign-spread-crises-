% ---------------- Motivation ----------------
\begin{frame}{Motivation: not all sovereign debt crises end with a default}
 \footnotesize
  \begin{itemize}
  \footnotesize
    \item \textbf{Sovereign debt crises are often resolved through restructuring}
    \begin{itemize}
    \item Sovereign default: (i) missed payments or a legal default, or (ii) a debt restructuring, resulting in creditor losses (``haircuts'') and/or coercion. \citep{ams2019sovereign}
    \end{itemize}

    \item \textbf{However, sovereign debt crisis can occur without a default}
    \begin{itemize}
     \footnotesize
    \item Eurozone debt crisis, crises in Mexico (1994--95), Thailand (1997--98), Brazil (1999, 2002), and Turkey (2001) did not result in external debt default or restructuring.
    \end{itemize}


    \item \textbf{A broader definition of debt crises has emerged}
    \begin{itemize}
 \footnotesize
    \item With the emergence of sovereign bond markets in EMDEs, debt crises can occur without default, involving elevated yields and rollover difficulties.

  \item The literature therefore uses broader measures based on bond market turbulence.


    \end{itemize}
  \end{itemize}
\end{frame}

% ---------------- This paper ----------------
\begin{frame}{This paper}

 \footnotesize
  \begin{itemize}

    \item \textbf{Research question:} What are the macroeconomic costs of sovereign debt crises with and without default, and does the pre-crisis sovereign-bank nexus amplify these effects?

    \item \textbf{Data:} JP Morgan EMBIG spread database
    \begin{itemize}
 \footnotesize
      \item 61 spread crisis episodes, 21 default-linked, 40 non-default
    \end{itemize}

    \item \textbf{Method:} Local projection with Augmented Inverse Probability
    Weighting (AIPW)

  \end{itemize}
\end{frame}

% ---------------- Preview of results ----------------

\begin{frame}{Preview of results}
 \footnotesize
  \begin{itemize}
    \item \textbf{Crisis type is predictable} (at onset): larger spread
    shocks, more trade openness, sharper depreciation $\to$ default;
    deeper bank credit, higher inflation $\to$ non-default

    \item \textbf{GDP and bank credit significantly differ by crisis type}
    \begin{itemize}
     \footnotesize
      \item GDP falls 1--2 pp following non-default crises vs.\ 4--9 pp
      following default-linked crises
      \item Bank credit turns positive (+7 pp) after non-default crises
      but contracts by more than 40 pp after default-linked crises
    \end{itemize}
    \item \textbf{Investment, FDI, and lending rates show no significant
    difference across types}
    \item \textbf{Bank claims on government rise sharply only after
    default-linked crises} ($\approx$+7 pp vs. $\approx$ -1 pp)

    \item \textbf{The sovereign-bank nexus matters, but only without
    default}
    \begin{itemize}
     \footnotesize
      \item High pre-crisis nexus is associated with a significantly
      higher GDP on impact in non-default crises (0.75 vs.\ $-$2.87 pp)
    \end{itemize}
  \end{itemize}
\end{frame}

% ---------------- Data and crisis classification ----------------
\begin{frame}{Identifying spread crisis episodes}
  \footnotesize
  \begin{itemize}
    \item \textbf{Identifying a crisis:} a country enters a spread
    crisis in the first year it clears either of two criteria -- its
    EMBIG spread crosses 1000 basis points, a level market
    participants treat as a ``psychological barrier''
    \citep{pescatori2007adequately}, or the spread jumps by at least
    378 basis points within the year (the HRT rate-of-increase rule)
    \item \textbf{Defining an episode:} consecutive distress years
    count as the same episode; a single tranquil year in between does
    not end it -- the episode is only considered over once two
    tranquil years follow in a row
    \item \textbf{Classifying episodes:} we check each episode
    against the \citet{asonuma2016sovereign} restructuring database
    (updated past 2020) -- if a restructuring happens during a spread
    crisis episode, we call it default-linked; otherwise, non-default
  \end{itemize}
\end{frame}

% ---------------- Main specification: overview ----------------
\begin{frame}{Main Specification: Probit model}
  \footnotesize

   We estimate each country-year's ex-ante probability of entering a
  non-default or default-linked spread crisis, and derive the
  predicted probability $\widehat{p}_{i,t+1}^S$
    \vspace{0.2em}

  \begin{equation}
  \Pr\!\left(D_{i,t+1}^S = 1 \mid X_{i,t}, Z_{i,t}\right) = \Phi\!\left(X_{i,t}\gamma^{S,X}, Z_{i,t}\gamma^{S,Z}\right)
  \end{equation}
  \vspace{-0.2em}
  \begin{itemize}
    \item $D_{i,t+1}^S \in \{0,1\}$: dummy equal to 1 if country $i$
    enters a crisis of type $S$ in year $t+1$
    \item $S \in \{$non-default, default-linked$\}$


    \item $X_{i,t}$: 8 baseline controls (GDP growth, public
    debt-to-GDP, banking crisis dummy, government expenditure-to-GDP,
    trade openness, bank credit-to-GDP, inflation, nominal exchange
    rate)
    \item $Z_{i,t}$: 3 predictors -- US federal funds rate, a
    contagion measure, years since the country's last default-linked
    onset
    \item $\gamma^{S,X}$, $\gamma^{S,Z}$: estimated coefficients on
    $X_{i,t}$ and $Z_{i,t}$, respectively, for crisis type $S$
  \end{itemize}
\end{frame}




\begin{frame}{Main Specification: OLS Local Projection}
  \footnotesize

  For each crisis type $S$ and horizon $h$, we estimate:

  \vspace{0.2em}
  \begin{equation}
  y_{i,t+h} - y_{i,t}
  =
  \alpha_i^{S,h}
  + \beta_D^{S,h} D_{i,t+1}^{S}
  + X_{i,t}'\beta_X^{S,h}
  + \mu_{i,t+h}^{S}
  \end{equation}
  \vspace{-0.2em}


 Then we predict the outcome under two counterfactual scenarios corresponding to the treated and untreated states:


  \begin{equation}
  \widehat m_{1}^{S,h}
  =
  \widehat\alpha_i^{S,h}
  +\widehat\beta_D^{S,h}
  +X_{i,t}'\widehat\beta_X^{S,h}
  \qquad (D=1)
  \end{equation}

  \begin{equation}
  \widehat m_{0}^{S,h}
  =
  \widehat\alpha_i^{S,h}
  +X_{i,t}'\widehat\beta_X^{S,h}
  \qquad (D=0)
  \end{equation}

\end{frame}




% ---------------- Main Specification: combining into AIPW ----------------
\begin{frame}{Main Specification: AIPW estimator}
  \footnotesize

Following \citet{jorda2016time} and using the results from Equations (1)--(4),
we define the AIPW estimator as:

  \begin{equation}
  \resizebox{0.92\linewidth}{!}{$\displaystyle
  \Lambda_h^S = \frac{1}{N}\sum_{i}\sum_{t}\Biggl\{
  \underbrace{\left[\frac{D_{i,t+1}^S(y_{i,t+h}-y_{i,t})}{\widehat{p}_{i,t+1}^S}
    - \frac{(1-D_{i,t+1}^S)(y_{i,t+h}-y_{i,t})}{1-\widehat{p}_{i,t+1}^S}\right]}_{\text{IPW term}}
  - \underbrace{\frac{D_{i,t+1}^S - \widehat{p}_{i,t+1}^S}{\widehat{p}_{i,t+1}^S(1-\widehat{p}_{i,t+1}^S)}
    \left[(1-\widehat{p}_{i,t+1}^S)\widehat{m}_1^{S,h} + \widehat{p}_{i,t+1}^S\widehat{m}_0^{S,h}\right]}_{\text{augmentation term}}
  \Biggr\}
  $}
  \end{equation}
  \vspace{-0.2em}
  \begin{itemize}
    \item $\Lambda_h^S$: the AIPW estimate -- the effect of crisis type
    $S$ on the outcome at horizon $h$, our object of interest

    \item Doubly robust: $\Lambda_h^S$ stays consistent if either the
    probit \emph{or} the outcome model is correctly specified, not both


    \item Differences between non-default and default-linked estimates
    are assessed with a paired bootstrap test (governing) and the
    \citet{clogg1995statistical} test (companion, assumes independence)
  \end{itemize}
\end{frame}

% ---------------- Nexus specification ----------------
\begin{frame}{Sovereign-bank nexus: specification}

 \footnotesize

   We interact the crisis-type dummy with
    the high-nexus dummy inside the same local-projection framework

  \begin{center}

    \begin{equation}
    y_{i,t+h} - y_{i,t} = \delta_i^{S,h}
      + \Gamma^{HN,h} D_{i,t}^{HN} D_{i,t+1}^{S}
      + \Gamma^{LN,h} \left(1 - D_{i,t}^{HN}\right) D_{i,t+1}^{S}
      + X_{i,t}\Gamma^{S,h} + e_{i,t+h}^{S}
    \end{equation}
  \end{center}

  \begin{itemize}

   \item Where $\Gamma^{HN,h}$, $\Gamma^{LN,h}$ are effect of a crisis of type $S$ when pre-crisis sovereign exposure is high or low.
  \end{itemize}
\end{frame}

\sectionroadmap{4}

% ---------------- Conclusion ----------------
\begin{frame}{Conclusion}
 \footnotesize
  \begin{itemize}
    \item Sovereign spread crises without default carry real but modest
    output costs, far smaller than default-linked crises
    \item The sovereign-bank nexus amplifies the recovery in non-default
    crises, consistent with a bank-lending channel that is only active when
    the sovereign avoids default
    \item No significant nexus effect within default-linked crises, though
    the point estimates are consistently more negative for high-nexus
    countries -- a null result given the limited sample of default-linked
    episodes, not evidence of no effect
  \end{itemize}
\end{frame}
